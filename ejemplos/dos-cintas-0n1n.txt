// Máquina de dos cintas que decide L = {0ⁿ1ⁿ | n ≥ 1}.
// δ: Q × Γ² → Q × Γ² × {L, R, S}²; δ(qᵢ, a₁, a₂) = (qⱼ, b₁, b₂, D₁, D₂). S deja la cabeza en su lugar.
// w va en la cinta 1 y la cinta 2 empieza en blanco. Los 0 se copian como x en la cinta 2
// y después cada 1 consume una x. Las combinaciones no definidas van a qreject.

[componentes]
M = (Q, Σ, Γ, δ, q1, qaccept, qreject)
Q = {q1, q2, q3, qaccept, qreject}
Σ = {0, 1}
Γ = {0, 1, x, $, ⊔}
cintas = 2

[transiciones]
// q1: marca el inicio de la cinta 2 con $
δ(q1, 0, ⊔) = (q2, 0, $, S, R)

// q2: por cada 0 de la cinta 1 escribe una x en la cinta 2
δ(q2, 0, ⊔) = (q2, 0, x, R, R)
δ(q2, 1, ⊔) = (q3, 1, ⊔, S, L)

// q3: cada 1 de la cinta 1 consume una x de la cinta 2
δ(q3, 1, x) = (q3, 1, x, R, L)
δ(q3, ⊔, $) = (qaccept, ⊔, $, R, R)

[entrada]
w = 0011
